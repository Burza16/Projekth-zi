1. adatbázis
create database autokereskedes
default character set utf8
collate utf8_hungarian_ci;

2. tábla
create table autok(
    id int primary key AUTO_INCREMENT,
    marka varchar(20) not null,
    modell varchar(20) not null,
    kivitel varchar(10) not null,
    evjarat int(4) not null, 
    ar double(9,2) not null,
    rendszam varchar(8) not null,
    klíma decimal not null,
    üzemanyag varchar(6) not null
    );
